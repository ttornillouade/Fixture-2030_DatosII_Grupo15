// ===== 5.3 a) Estado al padre, luego al hijo, y un solo %Save del padre =====
do ##class(Fixture.Demo).Limpiar()
set sede = ##class(Fixture.Demo).NuevaSede("S01", "Estadio Centenario", "Montevideo", "URY", 60235)
set e1 = ##class(Fixture.Demo).NuevoEquipo("E001", "Seleccion 001", "Pais 001", "CONMEBOL", "A", "URY", 11, .j)
set e2 = ##class(Fixture.Demo).NuevoEquipo("E002", "Seleccion 002", "Pais 002", "UEFA", "A", "ESP", 11, .j)
set p = ##class(Fixture.Partido).%New()
set p.Codigo = "P001", p.Fase = "grupos", p.FechaHora = "2030-06-08 16:00:00"
set p.Sede = sede, p.EquipoLocal = e1, p.EquipoVisitante = e2
set p.ArbitroPrincipal = ##class(Fixture.Demo).NuevoArbitro("A01", "FIFA", "ARG")
// Estado del padre:
set sc = p.Iniciar()  write "Iniciar: ", $SYSTEM.Status.IsOK(sc), "  estado del partido: ", p.Estado
// Estado del hijo, agregado al padre en memoria:
set sc = p.RegistrarEvento("GOL", 23, j("E001-J09"), .gol)  write "RegistrarEvento: ", $SYSTEM.Status.IsOK(sc), "  gol.Partido es p: ", (gol.Partido = p)
write "Ids antes de guardar -> partido: '", p.%Id(), "'  evento: '", gol.%Id(), "'  sede: '", sede.%Id(), "'"
// Un solo guardado, invocado sobre el padre:
set sc = p.%Save()  write "p.%Save(): ", $SYSTEM.Status.IsOK(sc)
write "Ids después -> partido: ", p.%Id(), "  evento: ", gol.%Id(), "  sede: ", sede.%Id(), "  equipos: ", e1.%Id(), ",", e2.%Id()
write "En disco -> evento: ", ##class(Fixture.Evento).%ExistsId(gol.%Id()), "  E002: ", ##class(Fixture.Equipo).CodigoIdxExists("E002"), "  jugadores: ", ##class(Fixture.Demo).Contar("Fixture.Jugador")

// ===== 5.3 b) Falla controlada: propiedad requerida (Licencia) omitida a propósito =====
set x = ##class(Fixture.Arbitro).%New()
set x.Nombre = "Sin", x.Apellido = "Licencia", x.Documento = "DOC-X", x.FechaNacimiento = $ZDATEH("1990-01-01", 3), x.Nacionalidad = "URY", x.Categoria = "FIFA"
set sc = x.%Save()  write "x.%Save(): ", $SYSTEM.Status.IsOK(sc), !  do $SYSTEM.Status.DisplayError(sc)
write "Árbitros en disco: ", ##class(Fixture.Demo).Contar("Fixture.Arbitro"), " (solo A01: el inválido no se guardó)"

// ===== 5.3 c) Las tablas proyectadas en SQL coinciden con los objetos =====
write "Objetos -> ", p.Codigo, " | ", p.Sede.Estadio, " | local ", p.EquipoLocal.Codigo, " | ", gol.Minuto, "' ", gol.Tipo, " ", gol.Jugador.Codigo, " | estado ", p.Estado
do ##class(Fixture.Demo).Consulta("SELECT p.Codigo, p.Sede->Estadio AS Sede, p.EquipoLocal->Codigo AS EqLocal, e.Minuto, e.Tipo, e.Jugador->Codigo AS Jugador, p.Estado FROM Fixture.Evento e JOIN Fixture.Partido p ON e.Partido = p.ID")
